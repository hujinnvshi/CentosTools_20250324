WITH date_range AS (
    -- 获取上个月的第一天和最后一天
    SELECT 
        date_trunc('month', current_date - interval '1 month')::date AS start_date,
        (date_trunc('month', current_date) - interval '1 day')::date AS end_date
),
all_dates AS (
    -- 生成上个月所有日期
    SELECT generate_series(start_date, end_date, '1 day'::interval)::date AS day
    FROM date_range
),
work_days AS (
    -- 标记日期类型
    SELECT 
        day,
        EXTRACT(DOW FROM day) AS day_of_week,
        CASE 
            WHEN h.holiday_name = '调休工作日' THEN '工作日'
            WHEN h.holiday_date IS NOT NULL THEN '节假日'
            WHEN EXTRACT(DOW FROM day) IN (0, 6) THEN '周末'
            ELSE '工作日'
        END AS day_type
    FROM all_dates d
    LEFT JOIN public_holidays h ON d.day = h.holiday_date
)
-- 计算工作天数
SELECT 
    COUNT(*) AS total_days,
    COUNT(*) FILTER (WHERE day_type = '工作日') AS work_days,
    COUNT(*) FILTER (WHERE day_type = '周末') AS weekend_days,
    COUNT(*) FILTER (WHERE day_type = '节假日') AS holiday_days
FROM work_days;


CREATE TABLE monthly_work_days (
    id SERIAL PRIMARY KEY,
    year INTEGER NOT NULL,
    month INTEGER NOT NULL CHECK (month BETWEEN 1 AND 12),
    work_days INTEGER NOT NULL CHECK (work_days BETWEEN 0 AND 31),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE (year, month)
);

-- 创建临时日期表
WITH months AS (
    SELECT generate_series(1, 12) AS month
),
all_months AS (
    SELECT 
        2025 AS year,
        month,
        make_date(2025, month, 1) AS start_date,
        (make_date(2025, month, 1) + interval '1 month' - interval '1 day')::date AS end_date
    FROM months
),
all_days AS (
    SELECT
        am.year,
        am.month,
        generate_series(am.start_date, am.end_date, '1 day'::interval)::date AS day
    FROM all_months am
),
work_day_calculation AS (
    SELECT
        ad.year,
        ad.month,
        ad.day,
        EXTRACT(DOW FROM ad.day) AS day_of_week,
        CASE 
            WHEN h.holiday_name = '调休工作日' THEN TRUE  -- 调休算工作日
            WHEN h.holiday_date IS NOT NULL THEN FALSE -- 节假日不算工作日
            WHEN EXTRACT(DOW FROM ad.day) IN (0, 6) THEN FALSE -- 周末不算工作日
            ELSE TRUE -- 其他日子算工作日
        END AS is_work_day
    FROM all_days ad
    LEFT JOIN public_holidays h ON ad.day = h.holiday_date
)
-- 插入2025年每月工作日数据
INSERT INTO monthly_work_days (year, month, work_days)
SELECT
    year,
    month,
    COUNT(*) FILTER (WHERE is_work_day) AS work_days
FROM work_day_calculation
GROUP BY year, month
ORDER BY year, month;
